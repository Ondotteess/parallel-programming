\section*{Task 1.1}

\subsection*{Sample 1}

\begin{itemize}
\item A запускает B
\item A делает \texttt{join(B)} и ждёт
\item B выбрасывает исключение и завершается
\item \texttt{join()} в A возвращается
\item Исключение из B в A не передаётся
\end{itemize}

\subsection*{Sample 2}

\begin{itemize}
\item A запускает B
\item B выбрасывает исключение и завершается
\item A дожидается завершения B и запускает C
\item C джоинит B, но B уже завершен -> C возвращается сразу
\item Исключение не передается ни в А ни В С
\end{itemize}

\subsection*{Sample 3}

\begin{itemize}
\item A запускает B
\item A делает джоинит B
\item B падает
\item \texttt{join()} в A возвращается
\item A завершается
\item D джоинит A, и продолжает выполнение
\end{itemize}
